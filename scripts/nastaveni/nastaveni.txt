# Který model Claude hodnocení píše.
#   claude-haiku-4-5-20251001 = levný a rychlý (výchozí)
#   claude-sonnet-4-6         = několikanásobně dražší, ale důkladnější
model = claude-haiku-4-5-20251001

# Kolik odkazů na zdroje se smí zobrazit u jednoho bodu. (1 až 5)
maximalne_zdroju = 3

# Jak dlouhý musí být doklad, aby stav „splněno" obstál. Když model
# neuvede aspoň tolik znaků konkrétního dokladu (číslo ve Sbírce
# zákonů, datum účinnosti), sníží se stav na „částečně splněno".
# Vyšší číslo = přísnější web. (0 až 200)
minimalni_delka_dokladu = 12

# Vyžadovat doklad i u stavu „porušeno". Obvinění z porušení slibu
# je stejně silné tvrzení jako tvrzení o splnění, ale až do srpna 2026
# si o doklad říkal prompt jen u „splněno" — a web tak vydával obvinění,
# která neměla o co se opřít. Když je zapnuto a doklad chybí, stav
# spadne na „jen deklarováno".
#
# ZAPÍNEJ AŽ POTÉ, co proběhl aspoň jeden běh s upraveným promptem
# a v logu je vidět, že model doklad u „porušeno" opravdu dodává.
# Zapnuto dřív by naráz srazilo všechny porušené sliby, protože jejich
# doklad je prázdný podle staré instrukce, ne vinou modelu.
#   0 = vypnuto (výchozí)
#   1 = zapnuto
latka_poruseno = 1

# Kolik znaků z minulotýdenního komentáře model uvidí jako kontext.
# Delší = víc souvislosti mezi týdny, ale dražší běh. (40 až 1000)
delka_predchoziho_komentare = 180

# Kolik týdenních snímků se uchovává pro graf vývoje. 52 = jeden rok.
# (4 až 520)
historie_tydnu = 52

# Kolik posledních stavů daného bodu se modelu ukáže v promptu.
# (0 až 20)
historie_v_promptu = 6

# Kolikrát smí model vyhledávat na webu při hodnocení jedné oblasti.
# Víc = důkladnější a dražší. Běh má 18 oblastí. (1 až 10)
vyhledavani_hodnoceni = 4

# Totéž pro sběr zpráv týdne — jeden dotaz za celý běh. (1 až 15)
# POZOR, vyšší číslo nepomáhá: při 12 model navrhl jen 3 zprávy z 9
# žádaných, při 8 dodal všech 7. Výsledky vyhledávání zaplní kontext
# a odpověď je pak kratší a častěji nevalidní.
vyhledavani_zpravy = 8

# Kolik zpráv se na webu zobrazí. (1 až 12)
pocet_zprav = 5

# O kolik zpráv navíc se modelu řekne. Část kandidátů vždy vypadne —
# vymyšlený odkaz, rozcestník místo článku, příliš stará zpráva, nebo
# druhá zpráva ze stejného webu (z jednoho webu bereme jen jednu).
# Když se na webu zobrazuje méně zpráv než pocet_zprav, zvyš tohle
# číslo; v logu běhu je vidět, kolik a proč jich vypadlo. (0 až 10)
zprav_navic_k_vyberu = 4

# Jak staré zprávy se ještě berou jako „zprávy týdne". (1 až 30)
zpravy_pocet_dni = 7

# Rezerva navíc, kdyby se páteční běh opozdil, a zároveň úleva pro
# případ, kdy za týden nebylo dost zpráv. Skutečná hranice je
# zpravy_pocet_dni + zpravy_tolerance_dni = 12 dní. (0 až 14)
zpravy_tolerance_dni = 5

# ==========================================================
#  JAZYKOVÁ KOREKTURA
# ==========================================================

# Po vyhodnocení projde druhý model hotové texty a opraví překlepy,
# pády, slovakismy a cizí slova. Fakta, čísla, data ani citace zákonů
# měnit nesmí — hlídá to kontrola v programu, která podezřelou opravu
# zahodí a ponechá původní text.
#
# Logická hodnota (ano/ne) v tomto souboru neexistuje, používá se 0/1.
#   0 = vypnuto (nic se neplatí; odstranění značek <cite> a překlad
#       syrových klíčů jako „in_progress" běží i tak, je zdarma)
#   1 = zapnuto
korektura = 1

# Který model korekturu dělá. MUSÍ být jiný a silnější než ten, který
# texty psal — model své vlastní slovakismy nevidí.
#   claude-sonnet-4-6 = doporučeno, ~0,60 $ za běh (výchozí)
#   claude-opus-4-8   = nejdůkladnější, ~0,95 $ za běh
#   claude-sonnet-5   = NEPOUŽÍVAT bez zásahu do programu: sám od sebe
#                       zapíná „přemýšlení", které se vejde do stejného
#                       limitu jako odpověď, a hrozí useknutí JSONu.
korektura_model = claude-sonnet-4-6

# Pojistka. Když model v jedné kapitole přepíše víc než tolik procent
# úryvků, zahodí se celá dávka a kapitola zůstane neopravená. Chrání
# před pokaženým promptem, po kterém by model začal texty přepisovat
# místo opravovat. V praxi se opravuje kolem 50 %. (0 až 100)
korektura_nejvic_zmen_procent = 80

# --- stabilizace hodnoceni (od 08/2026) -------------------------------------

# Kazdy prechod z/do stavu "splneno" nebo "poruseno" overi druhy, silnejsi

overovat_prechody = 1

# Model na overovani prechodu. Ma byt silnejsi nez hodnotici model - slabsi

# Kolik vyhledavani smi tydeni sken udalosti pouzit na jednu kapitolu.
